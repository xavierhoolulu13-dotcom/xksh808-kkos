const BIZ_TYPES = ["Local Service Business","E-commerce","SaaS / Software","Agency / Freelance","Real Estate","Restaurant / Food","Health & Wellness","Consulting","Retail","Other"];

export default function App() {
  const [form, setForm] = useState({ bizType: "Local Service Business", bizName: "", monthlyRevenue: "", funnel: "", painPoints: "" });
  const [phase, setPhase] = useState("intake"); // intake | scanning | results
  const [data, setData] = useState(null);
  const [open, setOpen] = useState({});
  const [toast, setToast] = useState(null);

  const showToast = (msg) => { setToast(msg); setTimeout(() => setToast(null), 2200); };

  const copy = (text) => { navigator.clipboard.writeText(text); showToast("COPIED TO CLIPBOARD"); };

  const toggle = (id) => setOpen(p => ({ ...p, [id]: !p[id] }));

  const scan = async () => {
    if (!form.bizType) return;
    setPhase("scanning");
    const prompt = `Business: ${form.bizName || "unnamed"} | Type: ${form.bizType} | Monthly Revenue: ${form.monthlyRevenue || "unknown"} | Funnel description: ${form.funnel || "not provided"} | Known pain points: ${form.painPoints || "none specified"}`;
    try {
      const res = await fetch("https://api.anthropic.com/v1/messages", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          model: "claude-sonnet-4-20250514",
          max_tokens: 1000,
          system: SYSTEM,
          messages: [{ role: "user", content: prompt }],
        }),
      });
      const raw = await res.json();
      const text = raw.content?.[0]?.text || "{}";
      const clean = text.replace(/```json|```/g, "").trim();
      const parsed = JSON.parse(clean);
      setData(parsed);
      setPhase("results");
    } catch (e) {
      setPhase("intake");
      showToast("SCAN ERROR — CHECK CONNECTION");
    }
  };

  const severityColor = (s) => s === "critical" ? "var(--red2)" : s === "warning" ? "var(--amber)" : "var(--green2)";

  return (
    <>
      <style>{S}</style>
      <div className="app">
        <header className="hdr">
          <div className="hdr-brand">
            <div className="brand-seal">808</div>
            <div>
              <div className="brand-title">Revenue Leak Scanner</div>
              <div className="brand-sub">808 MONEY PRINTER · FUNNEL FILLER · XKSH808</div>
            </div>
          </div>
          <div className="hdr-right">
            {phase === "scanning" && <div className="live-badge">● SCANNING</div>}
            {phase === "results" && <div style={{ fontFamily: "var(--mono)", fontSize: 8, color: "var(--red2)", letterSpacing: 2 }}>{data?.leaks_found} LEAKS FOUND</div>}
          </div>
        </header>

        <main className="main">

          {/* INTAKE */}
          {phase === "intake" && (
            <div className="intake">
              <div className="section-title">Funnel Diagnostics</div>
              <div className="section-sub">DESCRIBE YOUR BUSINESS — ARC-808 IDENTIFIES EVERY REVENUE LEAK</div>
              <div className="intake-grid">
                <div className="field">
                  <label>Business Name</label>
                  <input placeholder="e.g. Aloha Cleaning Co." value={form.bizName} onChange={e => setForm(p => ({ ...p, bizName: e.target.value }))} />
                </div>
                <div className="field">
                  <label>Business Type</label>
                  <select value={form.bizType} onChange={e => setForm(p => ({ ...p, bizType: e.target.value }))}>
                    {BIZ_TYPES.map(b => <option key={b}>{b}</option>)}
                  </select>
                </div>
                <div className="field">
                  <label>Monthly Revenue (approx)</label>
                  <input placeholder="e.g. $8,000/mo" value={form.monthlyRevenue} onChange={e => setForm(p => ({ ...p, monthlyRevenue: e.target.value }))} />
                </div>
                <div className="field">
                  <label>Current Funnel Stages</label>
                  <input placeholder="e.g. Ad → DM → Quote → Close → Upsell" value={form.funnel} onChange={e => setForm(p => ({ ...p, funnel: e.target.value }))} />
                </div>
                <div className="field full">
                  <label>Known Pain Points / Where You're Losing Deals</label>
                  <textarea placeholder="e.g. Leads ghost after quote, no referral system, slow follow-up, high cart abandon rate..." value={form.painPoints} onChange={e => setForm(p => ({ ...p, painPoints: e.target.value }))} />
                </div>
              </div>
              <button className="scan-btn" onClick={scan}>⬛ RUN LEAK SCAN</button>
            </div>
          )}

          {/* SCANNING */}
          {phase === "scanning" && (
            <div className="scanning">
              <div style={{ fontFamily: "var(--display)", fontSize: 28, letterSpacing: 4, color: "var(--red2)" }}>SCANNING FUNNEL</div>
              <div className="scan-bar" style={{ width: "100%", maxWidth: 400 }} />

                          <div className="leak-meta">
                            <span className="leak-impact">{leak.revenue_at_risk}</span>
                            <span className="leak-severity">{leak.severity}</span>
                          </div>
                        </div>
                        <div className={`leak-toggle ${open[leak.id] ? "open" : ""}`}>▼</div>
                      </div>

                      {open[leak.id] && (
                        <div className="leak-body">
                          <div className="leak-cause">{leak.root_cause}</div>
                          <div>
                            <div className="filler-label">⚡ FILLER AGENT — {leak.filler_agent?.name?.toUpperCase()}</div>
                            <div className="yaml-block">{leak.filler_agent?.yaml}</div>
                          </div>
                          <div className="leak-actions">
                            <button className="act-btn primary" onClick={() => copy(leak.filler_agent?.yaml || "")}>COPY AGENT YAML</button>
                            <button className="act-btn" onClick={() => copy(`LEAK: ${leak.title}\nSTAGE: ${leak.stage}\nIMPACT: ${leak.revenue_at_risk}\nFIX: Deploy ${leak.filler_agent?.name}`)}>COPY BRIEF</button>
                          </div>
                        </div>
                      )}
                    </div>
                  ))}
                </div>
              </div>

              {/* FOOTER STAMP */}
              <div style={{ textAlign: "center", padding: "20px 0 8px", borderTop: "1px solid var(--border)" }}>
                <div style={{ fontFamily: "var(--display)", fontSize: 14, letterSpacing: 6, color: "var(--muted)" }}>XKSH808 · 808 MONEY PRINTER · LEAK-808</div>
                <div style={{ fontFamily: "var(--mono)", fontSize: 7, color: "var(--border2)", letterSpacing: 3, marginTop: 4 }}>KONTROLD KHAOS · SYSTEM OPERATIONAL</div>
              </div>

            </div>
          )}

        </main>
      </div>
      {toast && <div className="toast">✓ {toast}</div>}
    </>
  );
